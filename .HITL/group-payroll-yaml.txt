severity:

    label: truth_or_control_failure
    ship_allowed: false
    description: >
      May misrepresent payroll, hide incompleteness, corrupt financial meaning,
      imply an unproven state, or permit unsafe action.

  P1:
    label: product_or_ux_failure
    ship_allowed: false
    description: >
      Financial truth may remain correct, but workflow, hierarchy,
      accessibility, interaction, or comprehension is materially deficient.

  P2:
    label: polish_deficiency
    ship_allowed: conditional
    description: >
      Does not change payroll truth or prevent task completion but lowers
      visual quality, refinement, consistency, or premium feel.
    waiver_required: true

  NOT_DONE:
